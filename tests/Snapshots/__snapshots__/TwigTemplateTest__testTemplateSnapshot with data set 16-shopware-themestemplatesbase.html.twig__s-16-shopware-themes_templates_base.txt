{# SCRIPTS: Optimiertes Loading #}
{# ============================================================ #}
{% block base_body_script %}
    {# Plugins deregistrieren basierend auf Theme Config #}
    <script>
        document.addEventListener('DOMContentLoaded', function() {
            if (window.PluginManager) {
                {% if theme_config('PerformanceTheme.config.disableDatePicker') %}
                // DatePicker deregistrieren (-115 KB)
                if (window.PluginManager.getPlugin('DatePicker')) {
                    window.PluginManager.deregister('DatePicker');
                }
                {% endif %}

                {% if theme_config('PerformanceTheme.config.disableImageZoom') %}
                // ImageZoom deregistrieren (-72 KB Hammer.js)
                if (window.PluginManager.getPlugin('ImageZoom')) {
                    window.PluginManager.deregister('ImageZoom');
                }
                {% endif %}
            }
        });
    </script>

    {# Parent Scripts laden (defer ist Standard) #}
    {{ parent() }}

    {# Async Plugin Loading für Slider #}
    {% if theme_config('PerformanceTheme.config.asyncSliderLoading') %}
    <script type="module">
        // Slider nur laden wenn im Viewport
        const sliders = document.querySelectorAll('[data-base-slider]');
        if (sliders.length > 0) {
            const observer = new IntersectionObserver((entries) => {
                entries.forEach(entry => {
                    if (entry.isIntersecting) {
                        import('{{ asset('bundles/storefront/js/slider.js', 'asset') }}')
                            .then(module => {
                                // Slider initialisieren
                                window.PluginManager.initializePlugin('BaseSlider', entry.target);
                            });
                        observer.unobserve(entry.target);
                    }
                });
            }, { rootMargin: '100px' });

            sliders.forEach(slider => observer.observe(slider));
        }
    </script>
    {% endif %}
{% endblock %}
