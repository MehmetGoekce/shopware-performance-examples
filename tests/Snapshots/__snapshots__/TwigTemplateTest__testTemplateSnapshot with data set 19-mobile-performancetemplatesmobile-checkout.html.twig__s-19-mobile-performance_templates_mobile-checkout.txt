{# JavaScript #}
<script>
document.addEventListener('DOMContentLoaded', function() {
    const checkout = document.getElementById('mobile-checkout');

    // Section toggling
    checkout.querySelectorAll('[data-toggle="section"]').forEach(header => {
        header.addEventListener('click', function() {
            const section = this.closest('.checkout-section');
            const content = section.querySelector('.section-content');
            const isExpanded = this.getAttribute('aria-expanded') === 'true';

            // Close all sections
            checkout.querySelectorAll('.section-header').forEach(h => {
                h.setAttribute('aria-expanded', 'false');
            });
            checkout.querySelectorAll('.section-content').forEach(c => {
                c.setAttribute('aria-hidden', 'true');
            });

            // Open clicked section
            if (!isExpanded) {
                this.setAttribute('aria-expanded', 'true');
                content.setAttribute('aria-hidden', 'false');
            }
        });
    });

    // Continue buttons
    checkout.querySelectorAll('[data-next]').forEach(btn => {
        btn.addEventListener('click', function() {
            const currentSection = this.closest('.checkout-section');
            const nextSectionName = this.dataset.next;
            const nextSection = checkout.querySelector(`[data-section="${nextSectionName}"]`);

            // Mark current as completed
            currentSection.dataset.completed = 'true';

            // Update summary
            updateSectionSummary(currentSection);

            // Collapse current, expand next
            currentSection.querySelector('.section-header').setAttribute('aria-expanded', 'false');
            currentSection.querySelector('.section-content').setAttribute('aria-hidden', 'true');

            nextSection.querySelector('.section-header').setAttribute('aria-expanded', 'true');
            nextSection.querySelector('.section-content').setAttribute('aria-hidden', 'false');

            // Scroll to next section
            nextSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
        });
    });

    // Guest/Login toggle
    const checkoutTypeInputs = checkout.querySelectorAll('[name="checkout_type"]');
    checkoutTypeInputs.forEach(input => {
        input.addEventListener('change', function() {
            const guestForm = document.getElementById('guest-form');
            const loginForm = document.getElementById('login-form');

            if (this.value === 'guest') {
                guestForm.classList.remove('hidden');
                loginForm.classList.add('hidden');
            } else {
                guestForm.classList.add('hidden');
                loginForm.classList.remove('hidden');
            }
        });
    });

    // Form submission
    document.getElementById('checkout-form').addEventListener('submit', function(e) {
        const submitBtn = document.getElementById('submit-order');
        submitBtn.querySelector('.btn-text').classList.add('hidden');
        submitBtn.querySelector('.btn-loading').classList.remove('hidden');
        submitBtn.disabled = true;
    });

    function updateSectionSummary(section) {
        const sectionName = section.dataset.section;
        const summary = section.querySelector('.section-summary');

        if (sectionName === 'personal') {
            const email = document.getElementById('email').value;
            summary.textContent = email;
        } else if (sectionName === 'shipping') {
            const city = document.getElementById('city').value;
            const zipcode = document.getElementById('zipcode').value;
            summary.textContent = `${zipcode} ${city}`;
        } else if (sectionName === 'payment') {
            const selected = section.querySelector('input[name="paymentMethodId"]:checked');
            if (selected) {
                const label = selected.closest('label').querySelector('.option-name');
                summary.textContent = label.textContent;
            }
        }
    }
});
</script>
